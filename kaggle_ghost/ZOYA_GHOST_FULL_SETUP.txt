GPU_NAME = torch.cuda.get_device_name(0)

# 2. MODEL LOADING
# ==========================================
turbo_id = "Tongyi-MAI/Z-Image-Turbo"
base_id = "Tongyi-MAI/Z-Image"
OFFLOAD_DIR = "/kaggle/working/offload"
os.makedirs(OFFLOAD_DIR, exist_ok=True)
MAX_MEMORY = {0: f"{max(12, int(VRAM_GB-2))}GiB", "cpu": "18GiB"}

current_engine = {"name": None, "t2i": None, "i2i": None, "inpaint": None}
current_lora = {"path": None, "weight": 0.85}

def unload_current_engine():
    if current_engine["name"] is not None:
        current_engine["t2i"] = None
        current_engine["i2i"] = None
        current_engine["inpaint"] = None
        current_engine["name"] = None
        current_lora["path"] = None
        gc.collect()
        torch.cuda.empty_cache()
        torch.cuda.ipc_collect()

def load_engine(name, lora_path=None, lora_weight=0.85):
    if current_engine["name"] == name and current_lora["path"] == lora_path:
        if lora_path and current_lora["weight"] != lora_weight:
            try:
                current_engine["t2i"].set_adapters(["pornmaster"], adapter_weights=[lora_weight])
                current_lora["weight"] = lora_weight
            except: pass
        return

    unload_current_engine()
    model_id = turbo_id if name == "turbo" else base_id
    print(f"[*] Loading {name} engine...")

    t2i = ZImagePipeline.from_pretrained(
        model_id, torch_dtype=DTYPE,
        low_cpu_mem_usage=True,
        device_map="balanced",
        offload_folder=OFFLOAD_DIR,
        offload_state_dict=True,
        max_memory=MAX_MEMORY,
    )
    shared = t2i.components
    i2i = ZImageImg2ImgPipeline(**shared)
    inpaint = ZImageInpaintPipeline(**shared)

    # ZImagePipeline (S3-DiT) — no VAE slicing/tiling (SD-specific)

    if lora_path and lora_path.strip():
        try:
            t2i.load_lora_weights(lora_path)
            current_lora["path"] = lora_path
            current_lora["weight"] = lora_weight
            print(f"[✓] LoRA loaded: {lora_path}")
        except Exception as e:
            print(f"[!] LoRA: {e}")

    current_engine.update({"name": name, "t2i": t2i, "i2i": i2i, "inpaint": inpaint})
    print(f"[*] {name} engine ready.")

load_engine("turbo")
print("[*] Ghost Protocol V3 Active.")

# 3. HELPERS
# ==========================================
def b64_to_pil(s):
    if not s: return None
    return Image.open(BytesIO(base64.b64decode(s.split(',')[-1]))).convert("RGB")

def pil_to_b64(img):
    b = BytesIO()
    img.save(b, format="PNG")
    return base64.b64encode(b.getvalue()).decode("utf-8")

def clear_vram():
    gc.collect(); torch.cuda.empty_cache(); torch.cuda.ipc_collect()

# ==========================================
# 4. MASTER API
# ==========================================
def zoya_gateway(task, prompt, neg_prompt, mode, width, height, seed,
                 image_b64, mask_b64, lora_path="", lora_weight=0.85):
    start = time.time()
    clear_vram()
    try:
        width, height, seed = int(width), int(height), int(seed)
        gen = torch.Generator("cuda").manual_seed(seed) if seed != -1 else None
        lp = lora_path.strip() if lora_path else ""
        lw = float(lora_weight)

        if task == "Remove_BG":
            if not image_b64: raise ValueError("Input image missing")
            result = rembg.remove(b64_to_pil(image_b64))
        else:
            if mode in ("Fast", "Pro"):
                load_engine("turbo", lp, lw)
                p = current_engine
                steps, guidance = (8, 0.0) if mode == "Fast" else (16, 0.0)
                neg = None
            else:
                load_engine("base", lp, lw)
                p = current_engine
                steps, guidance = 45, 4.5
                neg = neg_prompt if neg_prompt else "lowres, bad anatomy, bad hands, cropped, worst quality"

            img = b64_to_pil(image_b64) if image_b64 else None
            mask = b64_to_pil(mask_b64) if mask_b64 else None
            base_kw = {"prompt": prompt, "num_inference_steps": steps,
                       "guidance_scale": guidance, "generator": gen}
            if neg is not None: base_kw["negative_prompt"] = neg

            if task == "Text2Image":
                result = p["t2i"](**base_kw, width=width, height=height).images[0]
            elif task == "Image2Image":
                if not img: raise ValueError("Image missing")
                result = p["i2i"](**base_kw, image=img.resize((width, height)), strength=0.6).images[0]
            elif task == "Inpaint":
                if not img or not mask: raise ValueError("Image or Mask missing")
                result = p["inpaint"](**base_kw, image=img.resize((width, height)),
                                      mask_image=mask.resize((width, height)).convert("L"), strength=0.8).images[0]
            else:
                raise ValueError(f"Unknown task: {task}")

        t = round(time.time() - start, 2)
        return pil_to_b64(result), json.dumps({"status":"SUCCESS","task":task,"mode":mode,
                                                "time_seconds":t,"lora":lp if lp else "none"})
    except Exception as e:
        traceback.print_exc()
        return "", json.dumps({"status":"ERROR","task":task,"mode":mode,
                               "time_seconds":round(time.time()-start,2),"message":str(e)})
